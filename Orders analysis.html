import matplotlib.pyplot as plt
#Hàm A
def plot_pie(df, columns, palette='Set2', figsize=(13, 4)):
    fig, axes = plt.subplots(1, len(columns), figsize=figsize)
    if len(columns) == 1: axes = [axes]
    for ax, column in zip(axes, columns):
        counts = df[column].fillna('Unknown').value_counts()
        total = counts.sum()
        colors = plt.get_cmap(palette)(range(len(counts)))
        wedges, _, _ = ax.pie(counts.values, colors=colors, autopct=lambda pct: str(round(pct * total / 100)) if pct > 0 else '', startangle=90, counterclock=False, pctdistance=0.7, radius=0.8, textprops={'fontsize': 14}, wedgeprops={'edgecolor': 'white', 'linewidth': 1})
        legend_labels = [f'{value}: {count / total:.1%}' for value, count in counts.items()]
        ax.legend(wedges, legend_labels, title=column, loc='center left', bbox_to_anchor=(0.85, 0.5), fontsize=14, frameon=False)
        ax.set_title(f'Phân bố theo {column}', fontsize=14, fontweight='bold')
        ax.axis('equal')
    plt.tight_layout()
    plt.show()

import pandas as pd

def analyze_pies(df, columns):
    for column in columns:
        s = df[column].fillna('Unknown')
        counts = s.value_counts()
        pct = counts / counts.sum() * 100

        if column == 'Status':
            status = s.astype(str).str.strip().str.lower()
            returned = status.eq('returned').sum()
            returning = status.eq('returning').sum()
            total_return = returned + returning
            print(f'\nStatus:\nTổng hoàn hàng: {total_return}/{len(status)} đơn ({total_return/len(status):.1%}).\nReturned: {returned} đơn ({returned/len(status):.1%}).\nReturning: {returning} đơn ({returning/len(status):.1%}).')

        elif column == 'Fee paid to courier':
            fee = pd.to_numeric(df[column], errors='coerce')
            valid_fee = fee[fee > 0]
            print(f'\nFee paid to courier:\nPhí ship trung bình: {valid_fee.mean():.2f}, tính trên {valid_fee.count()} đơn có phí ship.')

        elif column == 'Quantity':
            top1, top2 = counts.index[0], counts.index[1]
            count1, count2 = counts.iloc[0], counts.iloc[1]
            pct1, pct2 = pct.iloc[0], pct.iloc[1]
            top2_pct = pct.iloc[:2].sum()
            concentration = 'tập trung cao' if pct1 >= 60 else 'tập trung vừa phải' if pct1 >= 40 else 'phân tán'
            print(f'\nQuantity:\nPhân phối {concentration}.\nNhóm {top1} lớn nhất với {count1} đơn ({pct1:.1f}%), tiếp theo là nhóm {top2} với {count2} đơn ({pct2:.1f}%).\nHai nhóm đầu chiếm tổng cộng {top2_pct:.1f}%.')

import matplotlib.pyplot as plt

def plot_hist(df,column,title):
    
    bins=20
    figsize=(10, 5)
    data = df[column].dropna()
    ax = data.plot(kind='hist',bins=bins,figsize=figsize,edgecolor='black')

    ax.set_xlabel('Return rate')
    ax.set_ylabel('Số lượng ' + title)
    ax.set_title('Phân phối Return rate của các ' + title)

    plt.tight_layout()
    plt.show()

def plot_profit_by_return_rate(df,return_col='return_rate',profit_col='profit',cod_col='cod', title=''):
    bins=20
    # Loại bỏ dòng thiếu dữ liệu
    data = df[[return_col, profit_col,cod_col]].dropna().copy()
    # Chia return_rate thành các khoảng
    data['return_rate_bin'] = pd.cut(data[return_col],bins=bins)
    # Tính tổng profit trong từng khoảng return_rate
    summary_by_bin = (data.groupby('return_rate_bin', observed=False)[[profit_col, cod_col]].sum())
    # Lấy trung điểm mỗi khoảng làm giá trị trục x
    x = [interval.mid for interval in summary_by_bin.index]
    # Xác định độ rộng của cả nhóm cột
    bin_width = (summary_by_bin.index[0].right- summary_by_bin.index[0].left)
    bar_width = bin_width * 0.4
    # Vẽ biểu đồ
    fig, ax = plt.subplots(figsize=(12, 5))
    ax.bar([value - bar_width / 2 for value in x],summary_by_bin[profit_col],width=bar_width,label='Tổng profit',edgecolor='black')
    ax.bar([value + bar_width / 2 for value in x],summary_by_bin[cod_col],width=bar_width,label='Tổng COD',edgecolor='black')
    ax.axhline(y=0,color='black',linewidth=1)
    ax.set_xlabel('Return rate')
    ax.set_ylabel('Tổng giá trị')
    ax.set_title('Tổng profit và COD theo Return rate'+ (f' của {title}' if title else ''))
    ax.legend()
    plt.tight_layout()
    plt.show()
